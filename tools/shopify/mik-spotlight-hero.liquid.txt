<script>
  (function () {
    var root = document.getElementById('MikSpot-{{ section.id }}');
    if (!root || root.dataset.ready) return;
    root.dataset.ready = '1';
    var img = root.querySelector('.mik-spot__lit');
    var fx = parseFloat(root.dataset.spotX) / 100, fy = parseFloat(root.dataset.spotY) / 100;
    var cur = null, target = null, raf = 0;
    var still = window.matchMedia('(prefers-reduced-motion: reduce)').matches;

    function home() {
      var r = img.getBoundingClientRect(), box = root.getBoundingClientRect();
      var iw = img.naturalWidth || 2200, ih = img.naturalHeight || 1375;
      var s = Math.max(r.width / iw, r.height / ih);
      var pos = getComputedStyle(img).objectPosition.split(' ').map(parseFloat);
      var ox = (r.width - iw * s) * (pos[0] / 100), oy = (r.height - ih * s) * (pos[1] / 100);
      return { x: r.left - box.left + ox + iw * s * fx, y: r.top - box.top + oy + ih * s * fy };
    }
    function paint(p) {
      root.style.setProperty('--mik-x', p.x.toFixed(1) + 'px');
      root.style.setProperty('--mik-y', p.y.toFixed(1) + 'px');
    }
    function tick() {
      var t = target || home();
      if (!cur || still) cur = { x: t.x, y: t.y };
      else { cur.x += (t.x - cur.x) * 0.14; cur.y += (t.y - cur.y) * 0.14; }
      paint(cur);
      raf = (Math.abs(t.x - cur.x) + Math.abs(t.y - cur.y) > 0.5) ? requestAnimationFrame(tick) : 0;
    }
    function kick() { if (!raf) raf = requestAnimationFrame(tick); }
    root.addEventListener('pointermove', function (e) {
      var b = root.getBoundingClientRect();
      target = { x: e.clientX - b.left, y: e.clientY - b.top };
      kick();
    });
    root.addEventListener('pointerleave', function () { target = null; kick(); });
    window.addEventListener('resize', function () { if (!target) { cur = null; kick(); } });
    if (img.complete) kick(); else img.addEventListener('load', function () { cur = null; kick(); });
  })();
</script>
